{{- /* func/count-words.html — count words the way a Chinese reader counts them. Hugo's .WordCount splits on spaces, so a 400-character Chinese paragraph scores as a handful of "words". This counts: - every CJK character (and CJK punctuation) as one - every run of non-space characters left over (Latin, numbers) as one Call with a Page, or with a plain-text string. Returns an int. */ -}} {{- $cjk := "[\\x{3000}-\\x{303F}\\x{3040}-\\x{30FF}\\x{3400}-\\x{4DBF}\\x{4E00}-\\x{9FFF}\\x{AC00}-\\x{D7AF}\\x{F900}-\\x{FAFF}\\x{FF00}-\\x{FFEF}]" -}} {{- $text := "" -}} {{- if eq "string" (printf "%T" .) -}} {{- $text = . -}} {{- else -}} {{- $text = .Plain -}} {{- end -}} {{- $text = replaceRE "(?s)<(script|style)[^>]*>.*?" " " $text -}} {{- $text = replaceRE "(?s)" " " $text -}} {{- $chars := len (findRE $cjk $text) -}} {{- $rest := replaceRE $cjk " " $text -}} {{- return (add $chars (len (findRE "[^\\s]+" $rest))) -}}
